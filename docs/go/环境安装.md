## 直接安装（简单省事）

**ubuntu下，直接输入go命令**，会提示找不到，并列出相关安装方法和版本，虽然不是最新，但是简单省事

## 一键安装脚本（可以安装到最新）

下面脚本中环境变量 GOPATH 是 Go 工具的安装目录，这样就可以直接运行工具了

```bash
#!/usr/bin/env bash

set -e

GO_PROFILE="/etc/profile.d/go.sh"
GOPATH_DIR="$HOME/go"
BASHRC="$HOME/.bashrc"


write_bashrc_env() {  
  # 删除旧的 go 相关 PATH（避免重复）
  sed -i '/# >>> Go environment >>>/,/# <<< Go environment <<</d' "$BASHRC"

  cat >> $BASHRC <<EOF

# >>> Go environment >>>
export PATH=\$PATH:/usr/local/go/bin
export GOPATH=\$HOME/go
export PATH=\$PATH:\$GOPATH/bin
# <<< Go environment <<<
EOF

}

# 获取最新 Go 版本
get_latest_version() {
  curl -s https://go.dev/VERSION?m=text
}

# 获取当前已安装版本
get_current_version() {
  go version 2>/dev/null | awk '{print $3}'
}

install_go() {
  echo "=== 获取最新 Go 版本 ==="
  LATEST_VERSION=get_current_version
  echo "最新版本: $LATEST_VERSION"

  DOWNLOAD_URL="https://go.dev/dl/${LATEST_VERSION}.linux-amd64.tar.gz"
  FILE_NAME="${LATEST_VERSION}.linux-amd64.tar.gz"

  echo "=== 下载 Go ==="
  wget -q --show-progress "$DOWNLOAD_URL"

  echo "=== 删除旧版本 /usr/local/go ==="
  sudo rm -rf /usr/local/go

  echo "=== 解压到 /usr/local ==="
  sudo tar -C /usr/local -xzf "$FILE_NAME"

  echo "=== 删除压缩包 ==="
  rm -f "$FILE_NAME"

  echo "=== 环境变量立即生效 ==="
  export PATH=$PATH:/usr/local/go/bin
  export GOPATH=$HOME/go
  export PATH=$PATH:$GOPATH/bin

  echo "=== 写入系统级环境变量到 $GO_PROFILE ==="
  sudo bash -c "cat > $GO_PROFILE" <<EOF
export PATH=\$PATH:/usr/local/go/bin
export GOPATH=\$HOME/go
export PATH=\$PATH:\$GOPATH/bin
EOF
  sudo chmod +x "$GO_PROFILE"

  echo "=== 写入用户级环境变量到 $BASHRC ==="
  write_bashrc_env

  echo "=== 创建 GOPATH 目录 ==="
  mkdir -p "$GOPATH_DIR/bin" "$GOPATH_DIR/src" "$GOPATH_DIR/pkg"

  echo "=== 配置 GOPROXY ==="
  go env -w GOPROXY=https://goproxy.cn,direct

  echo "=== Go 安装完成 ==="
  go version
}

update_go() {
  CURRENT_VERSION=$(get_current_version)
  LATEST_VERSION=$(get_latest_version)

  echo "当前版本: ${CURRENT_VERSION:-未安装}"
  echo "最新版本: $LATEST_VERSION"

  if [ "$CURRENT_VERSION" = "$LATEST_VERSION" ]; then
    echo "=== 已是最新版本，无需更新 ==="
    exit 0
  fi

  echo "=== 开始更新 Go ==="
  install_go
}

uninstall_go() {
  echo "=== 卸载 Go SDK ==="
  sudo rm -rf /usr/local/go

  echo "=== 删除系统环境变量 ==="
  sudo rm -f "$GO_PROFILE"

  echo "=== 清理 ~/.bashrc 中的 Go 配置 ==="
  sed -i '/# >>> Go environment >>>/,/# <<< Go environment <<</d' "$BASHRC" 2>/dev/null || true
  source "$BASHRC"

  echo "=== 清理 Go 缓存 ==="
  rm -rf "$HOME/.cache/go-build"
  rm -rf "$GOPATH_DIR/pkg/mod/cache" 2>/dev/null || true

  read -p "删除 GOPATH ($GOPATH_DIR)? (y/N): " CONFIRM
  if [[ "$CONFIRM" == "y" || "$CONFIRM" == "Y" ]]; then
    rm -rf "$GOPATH_DIR"
    echo "GOPATH 已删除"
  else
    echo "GOPATH 保留"
  fi

  echo "=== Go 已卸载 ==="
}

show_version() {
  CURRENT_VERSION=$(get_current_version)
  echo "当前 Go 版本: ${CURRENT_VERSION:-未安装}"
}

case "$1" in
  install|i)
    install_go
    ;;
  update|u)
    update_go
    ;;
  uninstall|rm)
    uninstall_go
    ;;
  version|v)
    show_version
    ;;
  *)
    echo "使用方法:"
    echo "  ./go-manager.sh install   # 安装"
    echo "  ./go-manager.sh update    # 更新"
    echo "  ./go-manager.sh uninstall # 卸载"
    echo "  ./go-manager.sh version   # 查看版本"
    ;;
esac
```
