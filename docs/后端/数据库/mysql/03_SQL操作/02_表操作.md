## 创建表

```sql
CREATE TABLE `user`
(
    `id`         BIGINT       NOT NULL AUTO_INCREMENT,
    `uid`        VARCHAR(64)  NOT NULL UNIQUE COMMENT '对外ID（ULID）',
    `created_at` DATETIME(6)  NOT NULL,
    `updated_at` DATETIME(6)  NOT NULL,

    `name`       VARCHAR(100) DEFAULT NULL COMMENT '真实姓名',
    `nickname`   VARCHAR(100) DEFAULT NULL COMMENT '昵称',
    `avatar`     VARCHAR(500) DEFAULT NULL COMMENT '头像地址',

    PRIMARY KEY (`id`)
) ENGINE = InnoDB
  DEFAULT CHARSET = utf8mb4
  COLLATE = utf8mb4_0900_ai_ci COMMENT = '用户主体';
```

## 查看表结构

```sql
DESC `user`;
SHOW CREATE TABLE `user`;
```

## 修改表

```sql
ALTER TABLE `user` ADD COLUMN `age` INT DEFAULT NULL COMMENT '年龄';
ALTER TABLE `user` MODIFY COLUMN `nickname` VARCHAR(50);
ALTER TABLE `user` DROP COLUMN `age`;
```

## 删除表

```sql
DROP TABLE IF EXISTS `user`;
```

## 重命名表

```sql
RENAME TABLE `user` TO `user_bak`;
-- 或
ALTER TABLE `user` RENAME TO `user_bak`;
```
