## 联合类型
1. 使用 typing.Union

```python
from typing import Union

def func(a: Union[int, str]) -> None:
    print(a)
```

2. 使用管道操作符 |

```python
def func(a: int | str) -> None:
    print(a)
```

## <font style="color:rgb(44, 44, 54);">字面量类型</font>
```python
from typing import Literal

def process_data(data: Literal["a", "b"]) -> None:
    if data == "a":
        print("Data is a")
    elif data == "b":
        print("Data is b")
    else:
        raise ValueError(f"Unexpected data: {data}")
```



## fastapi 返回类型支持泛型解决
```python
class ExportFormResponse(BaseModel):
    url: str
    ext: str


# FastAPI 无法识别 ApiResponse[ExportFormResponse] —— 所以这样包一层
class ExportFormApiResponse(ApiResponse[ExportFormResponse]):
    data: Optional[ExportFormResponse] = None


@router.post(
    "/export/form_excel",
    summary="导出表单(excel)",
    response_model=ExportFormApiResponse,
)
def export_form(data: ExportFormRequest) -> ApiResponse[ExportFormResponse]:

    exporter = FormExcelExporter(data.fields, data.form_data)
    wb = exporter.fill()

    saved_name = uuid.uuid4()
    temp_path = f"{UPLOAD_DIR}/{saved_name}.xlsx"
    wb.save(temp_path)

    return ApiResponse[ExportFormResponse].success(
        data=ExportFormResponse(url=FILE_SERVER_URL + str(saved_name), ext="xlsx"),
    )
```



## TypedDict、BaseModel
```python
from typing import NotRequired, Optional, TypedDict

from pydantic import BaseModel, Field


class FillFormField(TypedDict):
    id: str
    """id"""
    label: str
    type: str
    format: NotRequired[str]
    options: NotRequired[list[dict[str, str]]]
    children: NotRequired[list["FillFormField"]]
    group_name: NotRequired[str]


class FillFormUserFileFB(BaseModel):
    name: str
    path: str


class FillFormUserFile(TypedDict):
    name: str
    content: str


class FillFormFB(BaseModel):
    form_name: Optional[str] = None
    fields: list[FillFormField] = Field(..., description="需要填写的表单字段列表")
    user_input: str = Field(..., description="用户输入的内容")
    user_files: list[FillFormUserFileFB] = Field(..., description="用户上传的文件")


class FillFormContent(TypedDict):
    formName: str
    fields: list[FillFormField]
    userInput: str
    userFiles: list[FillFormUserFile]

# total 表示所有字段都可选
class FormWorkflowContentTD(TypedDict, total=False):
    """流程表单通知消息内容"""
    send_company_id: Optional[str]
    """公司id，null表示个人"""

    send_company_name: Optional[str]
    """公司名"""

    send_company_img: Optional[str]
    """公司logo"""

    receiver_id: str
    """消息接收人用户id"""

    send_time: float
    """UTC时间戳，需要根据时区转换回本地时间展示，单位秒"""

    msg_uuid: str
    """后端生成的uuid，格式：NTF-90a479f058564fbcbdad1ad07e25a4a2"""

    msg: FormWorkflowMsgTD
    """消息内容对象"""
```

## 断言
**使用 cast**：骗类型检查器

```python
cast(int, company_id)
```

**使用 assert**：会运行，会抛错，相当于if

```python
data = None
assert data is not None  # 这里会抛 AssertionError
```
